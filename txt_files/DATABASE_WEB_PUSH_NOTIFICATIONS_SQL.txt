-- WESCOMM Web Push Notification Subscriptions
-- Run this in Supabase SQL Editor once before testing phone/browser push notifications.

create table if not exists public.push_subscriptions (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references public.profiles(id) on delete cascade,
  endpoint text not null unique,
  endpoint_hash text,
  p256dh text not null,
  auth text not null,
  user_agent text,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  revoked_at timestamptz
);

create index if not exists push_subscriptions_user_id_idx
  on public.push_subscriptions(user_id);

create index if not exists push_subscriptions_active_user_id_idx
  on public.push_subscriptions(user_id)
  where revoked_at is null;

create unique index if not exists push_subscriptions_endpoint_hash_unique_idx
  on public.push_subscriptions(endpoint_hash)
  where endpoint_hash is not null;

alter table public.push_subscriptions enable row level security;

-- The backend uses the Supabase service role key, so normal browser users do not need
-- direct table access. These policies intentionally keep direct anon/auth access closed.
drop policy if exists "push_subscriptions_no_direct_select" on public.push_subscriptions;
drop policy if exists "push_subscriptions_no_direct_insert" on public.push_subscriptions;
drop policy if exists "push_subscriptions_no_direct_update" on public.push_subscriptions;
drop policy if exists "push_subscriptions_no_direct_delete" on public.push_subscriptions;

-- Verification query:
select
  n.nspname as schema_name,
  c.relname as table_name,
  c.relrowsecurity as row_level_security_enabled
from pg_class c
join pg_namespace n on n.oid = c.relnamespace
where n.nspname = 'public'
  and c.relname = 'push_subscriptions';
