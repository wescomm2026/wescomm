-- WESCOMM Student Reservation Offenses and Restrictions
-- Additive migration. Run once in the Supabase SQL Editor.

create extension if not exists pgcrypto;

alter type public.reservation_status add value if not exists 'NO_SHOW';
alter type public.inventory_movement_type add value if not exists 'RESERVATION_NO_SHOW';

do $$
begin
  create type public.student_offense_type as enum (
    'NO_SHOW',
    'LATE_CANCELLATION',
    'RESERVATION_SPAM'
  );
exception when duplicate_object then null;
end $$;

do $$
begin
  create type public.student_offense_status as enum ('ACTIVE', 'OVERTURNED');
exception when duplicate_object then null;
end $$;

do $$
begin
  create type public.restriction_status as enum ('ACTIVE', 'EXPIRED', 'LIFTED');
exception when duplicate_object then null;
end $$;

do $$
begin
  create type public.restriction_source as enum ('AUTOMATIC', 'MANUAL');
exception when duplicate_object then null;
end $$;

create table if not exists public.student_offenses (
  id uuid primary key default gen_random_uuid(),
  student_id uuid not null references public.profiles(id) on delete cascade,
  reservation_id uuid references public.reservations(id) on delete set null,
  type public.student_offense_type not null,
  status public.student_offense_status not null default 'ACTIVE',
  reason text not null check (char_length(trim(reason)) between 3 and 1000),
  occurred_at timestamptz not null default now(),
  confirmed_by_id uuid references public.profiles(id) on delete set null,
  overturned_by_id uuid references public.profiles(id) on delete set null,
  overturned_at timestamptz,
  overturn_reason text,
  created_at timestamptz not null default now()
);

create unique index if not exists student_offenses_reservation_type_unique_idx
  on public.student_offenses(reservation_id, type)
  where reservation_id is not null;

create index if not exists student_offenses_student_status_occurred_idx
  on public.student_offenses(student_id, status, occurred_at desc);

create table if not exists public.account_restrictions (
  id uuid primary key default gen_random_uuid(),
  student_id uuid not null references public.profiles(id) on delete cascade,
  offense_id uuid unique references public.student_offenses(id) on delete set null,
  level integer not null check (level between 1 and 3),
  source public.restriction_source not null,
  status public.restriction_status not null default 'ACTIVE',
  reason text not null check (char_length(trim(reason)) between 3 and 1000),
  starts_at timestamptz not null default now(),
  ends_at timestamptz,
  created_by_id uuid references public.profiles(id) on delete set null,
  lifted_by_id uuid references public.profiles(id) on delete set null,
  lifted_at timestamptz,
  lift_reason text,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  constraint account_restrictions_valid_window check (ends_at is null or ends_at > starts_at)
);

create index if not exists account_restrictions_student_status_ends_idx
  on public.account_restrictions(student_id, status, ends_at);

create index if not exists account_restrictions_source_level_idx
  on public.account_restrictions(source, level);

alter table public.student_offenses enable row level security;
alter table public.account_restrictions enable row level security;

-- The Express backend uses the service-role key and enforces student/staff/admin access.
-- No direct anon/authenticated policies are intentionally created for these tables.

select table_name
from information_schema.tables
where table_schema = 'public'
  and table_name in ('student_offenses', 'account_restrictions')
order by table_name;
