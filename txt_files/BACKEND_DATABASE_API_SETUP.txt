WESCOMM DATABASE SETUP GUIDE FOR GROUPMATES
===========================================

Purpose of this file:
This guide is ONLY for setting up the database side of WESCOMM.
No API route coding is included here.
No frontend implementation is included here.

Your groupmates can follow this guide to prepare:
- Supabase project
- PostgreSQL database tables
- database relationships
- row level security policies
- storage buckets
- realtime tables
- seed data

After they finish, they should send the required database connection details back to the frontend/backend implementer.


IMPORTANT RULES
===============

1. Do not send the Supabase service role key in a public group chat.
2. Do not commit .env.local to GitHub.
3. Use one Supabase project for WESCOMM.
4. Use the public schema for the tables below.
5. Use lowercase snake_case table and column names.
6. Enable Row Level Security on user-related tables.
7. Product/category data can be public read.
8. Student data, receipts, reservations, and messages must be protected.


DATABASE PLATFORM
=================

Use:

Supabase PostgreSQL

Supabase features needed:
- PostgreSQL database
- Authentication
- Storage
- Realtime
- SQL Editor


PART 1 - CREATE SUPABASE PROJECT
================================

1. Open Supabase.
2. Create a new project.
3. Suggested project name:

   wescomm

4. Choose a database password.
5. Save the password somewhere secure.
6. Choose a region close to the Philippines if available.
7. Wait for the project to finish provisioning.

PART 3 - DATABASE ENV VALUES NEEDED BY IMPLEMENTER
==================================================

The implementer will need these later:

NEXT_PUBLIC_SUPABASE_URL="https://YOUR_PROJECT_REF.supabase.co"
NEXT_PUBLIC_SUPABASE_ANON_KEY="YOUR_ANON_PUBLIC_KEY"
SUPABASE_SERVICE_ROLE_KEY="YOUR_SERVICE_ROLE_KEY"
DATABASE_URL="YOUR_TRANSACTION_POOLER_6543_CONNECTION_STRING_WITH_PGBOUNCER_TRUE_AND_TLS"
DIRECT_URL="YOUR_SESSION_OR_DIRECT_5432_CONNECTION_STRING_WITH_TLS"

Notes:

DATABASE_URL:
Used by the application during runtime. On Supabase/Vercel, use transaction
mode on port 6543 with pgbouncer=true and sslmode=require.

DIRECT_URL:
Used only for Prisma migration, logical backup, and restore. Use session/direct
mode on port 5432 with sslmode=require.

-- WESCOMM DATABASE SCHEMA
-- Run this inside Supabase SQL Editor.

create extension if not exists "pgcrypto";


-- ENUMS
--------

do $$
begin
  if not exists (select 1 from pg_type where typname = 'app_role') then
    create type public.app_role as enum ('STUDENT', 'STAFF', 'ADMIN');
  end if;

  if not exists (select 1 from pg_type where typname = 'product_status') then
    create type public.product_status as enum ('IN_STOCK', 'RESTOCK_SOON', 'OUT_OF_STOCK', 'ON_SALE');
  end if;

  if not exists (select 1 from pg_type where typname = 'inventory_movement_type') then
    create type public.inventory_movement_type as enum ('RESTOCK', 'SALE', 'RESERVATION_HOLD', 'RESERVATION_CANCEL', 'ADJUSTMENT');
  end if;

  if not exists (select 1 from pg_type where typname = 'reservation_status') then
    create type public.reservation_status as enum ('PENDING', 'CONFIRMED', 'READY_FOR_PICKUP', 'COMPLETED', 'CANCELLED');
  end if;

  if not exists (select 1 from pg_type where typname = 'payment_method') then
    create type public.payment_method as enum ('PAY_AT_COMMISSARY', 'E_WALLET_AT_PICKUP', 'CASH', 'GCASH');
  end if;

  if not exists (select 1 from pg_type where typname = 'receipt_status') then
    create type public.receipt_status as enum ('PENDING', 'VERIFIED', 'VOIDED');
  end if;

  if not exists (select 1 from pg_type where typname = 'notification_type') then
    create type public.notification_type as enum ('RESERVATION', 'RECEIPT', 'LOW_STOCK', 'MESSAGE', 'SYSTEM');
  end if;

  if not exists (select 1 from pg_type where typname = 'conversation_status') then
    create type public.conversation_status as enum ('OPEN', 'RESOLVED');
  end if;
end $$;


-- HELPER FUNCTIONS
-------------------

create or replace function public.set_updated_at()
returns trigger
language plpgsql
as $$
begin
  new.updated_at = now();
  return new;
end;
$$;

create or replace function public.current_user_role()
returns public.app_role
language sql
stable
security definer
set search_path = public
as $$
  select role
  from public.profiles
  where id = auth.uid()
  limit 1;
$$;

create or replace function public.is_staff_or_admin()
returns boolean
language sql
stable
security definer
set search_path = public
as $$
  select exists (
    select 1
    from public.profiles
    where id = auth.uid()
      and role in ('STAFF', 'ADMIN')
  );
$$;

create or replace function public.is_admin()
returns boolean
language sql
stable
security definer
set search_path = public
as $$
  select exists (
    select 1
    from public.profiles
    where id = auth.uid()
      and role = 'ADMIN'
  );
$$;


-- TABLES
---------

create table if not exists public.profiles (
  id uuid primary key references auth.users(id) on delete cascade,
  full_name text not null default '',
  email text not null unique,
  student_number text unique,
  phone text,
  department text,
  address text,
  role public.app_role not null default 'STUDENT',
  avatar_url text,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create table if not exists public.categories (
  id uuid primary key default gen_random_uuid(),
  name text not null unique,
  slug text not null unique,
  icon_url text,
  is_active boolean not null default true,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create table if not exists public.products (
  id uuid primary key default gen_random_uuid(),
  category_id uuid not null references public.categories(id) on delete restrict,
  name text not null,
  description text,
  image_url text,
  price numeric(12,2) not null default 0,
  old_price numeric(12,2),
  status public.product_status not null default 'IN_STOCK',
  stock integer not null default 0,
  low_stock_threshold integer not null default 10,
  is_active boolean not null default true,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  constraint products_stock_nonnegative check (stock >= 0),
  constraint products_price_nonnegative check (price >= 0)
);

create table if not exists public.product_variants (
  id uuid primary key default gen_random_uuid(),
  product_id uuid not null references public.products(id) on delete cascade,
  option_name text not null,
  option_value text not null,
  stock integer not null default 0,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  constraint product_variants_stock_nonnegative check (stock >= 0),
  unique (product_id, option_name, option_value)
);

create table if not exists public.inventory_movements (
  id uuid primary key default gen_random_uuid(),
  product_id uuid not null references public.products(id) on delete cascade,
  variant_id uuid references public.product_variants(id) on delete set null,
  type public.inventory_movement_type not null,
  quantity integer not null,
  previous_stock integer not null,
  new_stock integer not null,
  performed_by_id uuid references public.profiles(id) on delete set null,
  notes text,
  created_at timestamptz not null default now()
);

create table if not exists public.reservations (
  id uuid primary key default gen_random_uuid(),
  student_id uuid not null references public.profiles(id) on delete cascade,
  reference_code text not null unique,
  status public.reservation_status not null default 'PENDING',
  pickup_start timestamptz,
  pickup_end timestamptz,
  payment_method public.payment_method not null default 'PAY_AT_COMMISSARY',
  total_amount numeric(12,2) not null default 0,
  staff_notes text,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  constraint reservations_total_nonnegative check (total_amount >= 0)
);

create table if not exists public.reservation_items (
  id uuid primary key default gen_random_uuid(),
  reservation_id uuid not null references public.reservations(id) on delete cascade,
  product_id uuid not null references public.products(id) on delete restrict,
  variant_summary text,
  quantity integer not null,
  unit_price numeric(12,2) not null,
  subtotal numeric(12,2) not null,
  created_at timestamptz not null default now(),
  constraint reservation_items_quantity_positive check (quantity > 0),
  constraint reservation_items_unit_price_nonnegative check (unit_price >= 0),
  constraint reservation_items_subtotal_nonnegative check (subtotal >= 0)
);

create table if not exists public.receipts (
  id uuid primary key default gen_random_uuid(),
  receipt_code text not null unique,
  student_id uuid not null references public.profiles(id) on delete cascade,
  reservation_id uuid references public.reservations(id) on delete set null,
  total_amount numeric(12,2) not null default 0,
  payment_method public.payment_method not null default 'CASH',
  status public.receipt_status not null default 'PENDING',
  verification_hash text not null unique,
  receipt_image_url text,
  receipt_pdf_url text,
  issued_by_id uuid references public.profiles(id) on delete set null,
  issued_at timestamptz not null default now(),
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  constraint receipts_total_nonnegative check (total_amount >= 0)
);

create table if not exists public.notifications (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references public.profiles(id) on delete cascade,
  title text not null,
  message text not null,
  type public.notification_type not null default 'SYSTEM',
  read_at timestamptz,
  created_at timestamptz not null default now()
);

create table if not exists public.conversations (
  id uuid primary key default gen_random_uuid(),
  student_id uuid not null references public.profiles(id) on delete cascade,
  assigned_staff_id uuid references public.profiles(id) on delete set null,
  subject text not null,
  status public.conversation_status not null default 'OPEN',
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create table if not exists public.conversation_messages (
  id uuid primary key default gen_random_uuid(),
  conversation_id uuid not null references public.conversations(id) on delete cascade,
  sender_id uuid not null references public.profiles(id) on delete cascade,
  message text not null,
  created_at timestamptz not null default now()
);

create table if not exists public.faqs (
  id uuid primary key default gen_random_uuid(),
  question text not null,
  answer text not null,
  category text,
  is_published boolean not null default true,
  updated_by_id uuid references public.profiles(id) on delete set null,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create table if not exists public.app_settings (
  id uuid primary key default gen_random_uuid(),
  key text not null unique,
  value jsonb not null default '{}'::jsonb,
  updated_by_id uuid references public.profiles(id) on delete set null,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

-- AUTH PROFILE TRIGGER
-----------------------

create or replace function public.handle_new_user()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
begin
  insert into public.profiles (id, full_name, email, role)
  values (
    new.id,
    coalesce(new.raw_user_meta_data->>'full_name', new.raw_user_meta_data->>'name', ''),
    coalesce(new.email, ''),
    'STUDENT'
  )
  on conflict (id) do nothing;

  return new;
end;
$$;

drop trigger if exists on_auth_user_created on auth.users;
create trigger on_auth_user_created
after insert on auth.users
for each row execute function public.handle_new_user();

-- BACKEND-ONLY PUBLIC DATA ACCESS
----------------------------------
-- Supabase is used client-side for Auth only. All public-schema reads and
-- writes pass through the WESCOMM backend. Never expose the service-role key.

revoke all privileges on all tables in schema public from anon, authenticated;
revoke all privileges on all sequences in schema public from anon, authenticated;
revoke execute on all functions in schema public from anon, authenticated;
revoke all privileges on all tables in schema public from public;
revoke all privileges on all sequences in schema public from public;
revoke execute on all functions in schema public from public;

-- These defaults apply to the current WESCOMM migration owner. Supabase's
-- platform-managed supabase_admin defaults are intentionally not modified.
alter default privileges in schema public
revoke all privileges on tables from anon, authenticated;
alter default privileges in schema public
revoke all privileges on sequences from anon, authenticated;
alter default privileges in schema public
revoke execute on functions from anon, authenticated;
alter default privileges in schema public
revoke all privileges on tables from public;
alter default privileges in schema public
revoke all privileges on sequences from public;
-- Function EXECUTE is granted to PUBLIC globally by PostgreSQL, so this revoke
-- must also be global for the current migration owner.
alter default privileges
revoke execute on functions from public;

-- Preserve the explicit server-only service-role boundary.
grant all privileges on all tables in schema public to service_role;
grant all privileges on all sequences in schema public to service_role;
grant execute on all functions in schema public to service_role;
alter default privileges in schema public
grant all privileges on tables to service_role;
alter default privileges in schema public
grant all privileges on sequences to service_role;
alter default privileges in schema public
grant execute on functions to service_role;

-- Prisma migration history is internal metadata. The table owner keeps access,
-- while PostgREST roles have no policies or grants on it.
do $block$
begin
  if to_regclass('public._prisma_migrations') is not null then
    alter table public._prisma_migrations enable row level security;
    revoke all privileges on table public._prisma_migrations from public;
    revoke all privileges on table public._prisma_migrations from anon, authenticated;
    revoke all privileges on table public._prisma_migrations from service_role;
  end if;
end
$block$;

PART 15 - COMMON ERRORS
=======================

Error:
relation "public.profiles" does not exist

Fix:
Run the full schema SQL from the top.


Error:
type "public.app_role" does not exist

Fix:
The enum creation part did not run.
Run the enum section again.


Error:
permission denied for schema storage

Fix:
Create buckets from Supabase Dashboard instead of SQL.
Then apply policies from SQL Editor.


Error:
cannot add table to publication

Fix:
The table may already be added to realtime.
Check Database -> Replication.


Error:
new row violates row-level security policy

Fix:
Check if the user is logged in.
Check if the profile row exists.
Check if the user has the correct role.
