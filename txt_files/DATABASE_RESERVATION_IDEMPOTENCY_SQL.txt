-- WESCOMM Reservation Checkout Idempotency
-- Additive migration. Run once in the Supabase SQL Editor before testing checkout.

create extension if not exists pgcrypto;

create table if not exists public.reservation_idempotency_keys (
  id uuid primary key default gen_random_uuid(),
  student_id uuid not null references public.profiles(id) on delete cascade,
  idempotency_key text not null,
  request_hash text not null,
  reservation_id uuid unique references public.reservations(id) on delete set null,
  expires_at timestamptz not null,
  created_at timestamptz not null default now(),
  constraint reservation_idempotency_key_length check (char_length(idempotency_key) between 16 and 128),
  constraint reservation_idempotency_hash_length check (char_length(request_hash) = 64),
  constraint reservation_idempotency_expiry check (expires_at > created_at)
);

create unique index if not exists reservation_idempotency_student_key_unique_idx
  on public.reservation_idempotency_keys(student_id, idempotency_key);

create index if not exists reservation_idempotency_expires_at_idx
  on public.reservation_idempotency_keys(expires_at);

alter table public.reservation_idempotency_keys enable row level security;

-- Only the Express backend service-role client may access these records directly.
-- Browser clients intentionally receive no direct table policy.

select
  c.relname as table_name,
  c.relrowsecurity as row_level_security_enabled
from pg_class c
join pg_namespace n on n.oid = c.relnamespace
where n.nspname = 'public'
  and c.relname = 'reservation_idempotency_keys';
